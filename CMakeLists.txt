cmake_minimum_required(VERSION 3.18)
project(exHadClient LANGUAGES CXX)
add_library(exhad_client INTERFACE)
add_library(exhad::client ALIAS exhad_client)
target_include_directories(exhad_client INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_features(exhad_client INTERFACE cxx_std_17)
option(EXHAD_BUILD_EXAMPLE "Build the host-Pythia integration example" OFF)
if(EXHAD_BUILD_EXAMPLE)
  set(PYTHIA8_DIR "" CACHE PATH "Host Pythia installation")
  find_path(PYTHIA8_INCLUDE Pythia8/Pythia.h HINTS ${PYTHIA8_DIR}/include REQUIRED)
  find_library(PYTHIA8_LIBRARY pythia8 HINTS ${PYTHIA8_DIR}/lib REQUIRED)
  add_executable(pythia8_decays examples/pythia8_decays.cc)
  target_include_directories(pythia8_decays PRIVATE ${PYTHIA8_INCLUDE})
  target_link_libraries(pythia8_decays PRIVATE exhad::client ${PYTHIA8_LIBRARY})
endif()
